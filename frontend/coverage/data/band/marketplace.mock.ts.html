function buildSeptember(bookedDays: number[]): VenueAvailabilityDay[] {
  return Array.from({ length: 30 }, (_, index) => {
    const day = index + 1;
    return {
      date: `2026-09-${String(day).padStart(2, "0")}`,
      status: bookedDays.includes(day) ? ("Booked" as const) : ("Available" as const),
    };
  });
}
